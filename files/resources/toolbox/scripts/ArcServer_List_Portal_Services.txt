import arcgis, arcpy
from arcgis.gis import GIS
# Gets services from your AGOL or ArcGIS Portal
# I put this together from https://support.esri.com/en/technical-article/000016853 it pretty much walks you through it 

user = "you@where" #Your username
password = "p33kAb00!" #Password
url = "https://gis.sometown.somecoutnry/someserver/home/" #server url
gis = GIS(url, user, password) #The function where this stuff is passed to access the hosting server
print("Login successful") #Print statement to let you know you got this far

try: #Try except to capture error
    fcs = gis.content.search("owner:" + "*", item_type="Map Service", max_items=100) #fcs should be the list being created the "*" will grab all owners add your username to get only what you own. Also 500 seems to be the max items to return I get an error when I ask for more.
    for fc in fcs: #Loops through a list and prints an entity as a new line in the command console
        try: #Incase it hits something it cannot print it will skip it and go to the next item.
            print(str(fc.title)+" "+str(fc.url)+" "+str(fc.type)) #Prints the entity
        except: #If the Print fails this is the continue logic, a print statement can be put here to let you know if something was skipped.
            continue

except Exception: #Error capture code.
    e = arcpy.sys.exc_info()[1]
    print(e.args[0])
