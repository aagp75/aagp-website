#Start
#Import Modules
print("Import Modules")
import arcpy
import os
import sys
import time

print(time.time)
# This is an oddball but there is a lot of functions going on so might be worth looking at. Basically I had 2 filegeodatabases with matching schemes I needed to combine but using copy paste duplicated domains, fields
# and other components I didn't want. So I wrote this to go through and check things, remove duplications, and reassign domains to to fields that got changed. It is very specific to the databases I was working with at
# the time.
#Setting Workspaces
print("Set Workspaces")
arcpy.env.workspace = r"k:\Ctown_Model\test"
SourceCMDM = r"k:\Ctown_Model\CTown_CMDM.gdb"
SourceWUNM = r"k:\Ctown_Model\CTown_WUNM.gdb"
wunm = os.path.join(r"k:\Ctown_Model\test","WUNM.gdb")
cmdm = os.path.join(r"k:\Ctown_Model\test","CMDM.gdb")
ddb = os.path.join(r"k:\Ctown_Model\test","Domains.gdb")

#Setting Variables
sTime = time.time()
print("Set Variables")
arcpy.env.workspace = wunm
dsList = arcpy.ListDatasets()
dListw = arcpy.da.ListDomains(wunm) #List of Water and Sewer DB Domains
print(dsList)
expLoc = r"k:\Ctown_Model\test\exports"

#Begin Script

#Step 1 Copy DB
arcpy.env.workspace = r"k:\Ctown_Model\test" #Reset Workspace
arcpy.env.overwriteOutput = True #Overwrite Output
try: #Attempt the Copy
    print("Copying databases...")
    arcpy.Copy_management(SourceCMDM, cmdm)
    arcpy.Copy_management(SourceWUNM, wunm)
except Exception: #Error Capture
    print("Copy DB Error")
    e = arcpy.sys.exc_info()[1]
    print(e.args[0])
    arcpy.ClearWorkspaceCache_management()
    sys.exit()
print("Databases Copied")
arcpy.management.Delete(os.path.join(cmdm, "WaterDistribution", "wAbandonedLine"))
arcpy.management.Delete(os.path.join(cmdm, "WaterDistribution", "wAbandonedPoint"))
print("Deleted PITA Featureclasses")

arcpy.ClearWorkspaceCache_management()

Time1 = ((time.time() - sTime))
print("Step 1 Complete --- %s seconds --- " % (Time1))

#Step 2 Delete Utility Datasets
arcpy.env.workspace = wunm
for ds in dsList:
    arcpy.env.workspace = (os.path.join(cmdm))
    try:
        print("Delete "+str(ds))
        arcpy.management.Delete(str(ds))
    except Exception:
        print("Delete Error " + str(ds))
        e = arcpy.sys.exc_info()[1]
        print(e.args[0])
        arcpy.ClearWorkspaceCache_management()
        sys.exit()
arcpy.ClearWorkspaceCache_management()
Time2 = ((time.time() - Time1))
print("Step 2 Complete --- %s seconds --- " % (Time2))

#Step 3 Build New Datasets
arcpy.env.workspace = wunm
for ds in dsList:
    arcpy.env.workspace = (os.path.join(cmdm))
    try:
        arcpy.management.CreateFeatureDataset(arcpy.env.workspace, str(ds), os.path.join(wunm,ds))
    except Exception:
        print("Delete Error " + str(ds))
        e = arcpy.sys.exc_info()[1]
        print(e.args[0])
        arcpy.ClearWorkspaceCache_management()
        sys.exit()
arcpy.ClearWorkspaceCache_management()
Time3 = ((time.time() - Time2))
print("Step 3 Complete --- %s seconds --- " % (Time3))

#Step 4 Check and Setup Domains
for dom in dListw:
    print(dom.name)
    arcpy.DomainToTable_management(os.path.join(arcpy.env.workspace, wunm), str(dom.name), os.path.join(ddb, str(dom.name)), "CODE", "DESC", "")
print("Domains exported to tables")

print("Starting Domain Inspection and Transfer")
for dw in dListw: #Loop through Water and Sewer DB Domains
    print("Current "+str(dw.name))
    dnamesc = [domain.name for domain in arcpy.da.ListDomains(cmdm)] #Domain list for existing CMDM Domains
    if dw.name in dnamesc: #Check if Water and Sewer Domain exists in CMDM
        print(str(dw.name)+" found")
        if dw.domainType == "Range": #Check if Domain is Range
            print("Skipping Range Domain "+str(dw.name))
            continue #Skipping Ranged Domain
        else:
            print("Updating Domain "+str(dw.name))
            arcpy.management.TableToDomain(os.path.join(ddb, dw.name), "Code", "Desc", os.path.join(cmdm), dw.name, dw.description, "REPLACE") #Update the Domain with values from Water Utility DB
    else:
        print(str(dw.name)+" not found, creating")
        arcpy.management.TableToDomain(os.path.join(ddb, dw.name), "Code", "Desc", os.path.join(cmdm), dw.name, dw.description, "REPLACE") #If none exists create
print("All Domains Transfered")
arcpy.ClearWorkspaceCache_management()
Time4 = ((time.time() - Time3))
print("Step 4 Complete --- %s seconds --- " % (Time4))

#Step 5 Check and Setup Featureclasses
arcpy.env.workspace = r"k:\Ctown_Model\test" #Reset Workspace

for ds in dsList:
    arcpy.env.workspace = os.path.join(wunm,ds)
    print(str(ds)+" Current Workspace in "+str(wunm))
    fcListWUNM = arcpy.ListFeatureClasses()
    arcpy.env.workspace = os.path.join(cmdm, ds)
    fcListCMDM = arcpy.ListFeatureClasses()
    for fc in fcListWUNM:
        if fc in fcListCMDM:
            continue
        else:
            fcd = arcpy.Describe(os.path.join(wunm, ds, fc))
            print("Found missing Featureclass "+str(fc)+ " adding Featureclass to CMDM")
            arcpy.management.CreateFeatureclass(arcpy.env.workspace, str(fc), (str(fcd.shapeType).upper()), os.path.join(wunm, ds, fc), "SAME_AS_TEMPLATE", "SAME_AS_TEMPLATE", "", "", "", "", "", "")
            arcpy.ClearWorkspaceCache_management()
            fields = arcpy.ListFields(os.path.join(arcpy.env.workspace, fc))
            print("Checking for and removing copied domains")
            for field in fields:
                if str(field.domain) != "":
                    arcpy.RemoveDomainFromField_management(os.path.join(arcpy.env.workspace,fc), str(field.name))
domListCMDM = arcpy.da.ListDomains(cmdm)
print("Checking for and removing duplicate domains from the database")
for dom in domListCMDM:
    if str(dom.name)[2:-1] == "_1":
        print("Deleting "+str(dom.name))
        arcpy.management.DeleteDomain(arcpy.env.workspace, str(dom.name))
    else:
        continue
arcpy.ClearWorkspaceCache_management()

Time5 = ((time.time() - Time4))
print("Step 5 Complete --- %s seconds --- " % (Time5))

#Step 6 Check and Setup Fields
arcpy.env.workspace = r"k:\Ctown_Model\test" #Reset Workspace

for ds in dsList:
    arcpy.env.workspace = os.path.join(wunm,ds)
    print(str(ds)+" Current Workspace in "+str(wunm))
    fcList = arcpy.ListFeatureClasses()
    for fc in fcList:
        arcpy.env.workspace = os.path.join(wunm, ds, fc)
        print(str(fc) + " Current Featureclass in"+str(ds))
        fields = arcpy.ListFields(arcpy.env.workspace) #Building a list of fields for the source featureclass
        for field in fields: #Looping through fields in the source featureclass
            print("Current Field "+str(field.name)+" in "+str(fc))
            fieldnames = [field.name for field in arcpy.ListFields(os.path.join(cmdm, ds, fc))] #Building a reference list of fields from the target featureclass
            if field.name in fieldnames: #Checking if field exists in target featureclass
                print(str(field.name)+" found in CMDM Database")
                if str(field.type)== "Geometry": #Check and skip OBJECTID
                    print(str(field.name)+" skipped, Geometry field")
                    continue
                if str(field.type)== "GlobalID": #Check and skip SHAPE
                    print(str(field.name) + " skipped, GlobalID field")
                    continue
                if str(field.type)== "OID": #Check and skip GlobalID
                    print(str(field.name) + " skipped, OID field")
                    continue
                if str(field.name)== "ENABLED": #Check and skip ENABLED
                    print(str(field.name) + " skipped, Enabled field")
                    continue
                else: #If none of the above fileds continue to inspect field
                        sourcef = arcpy.ListFields(os.path.join(wunm, ds, fc), field.name) #Set source location
                        syncf = arcpy.ListFields(os.path.join(cmdm, ds, fc), field.name) #Set target location
                        if syncf[0].domain == sourcef[0].domain: #Compre the the source and target fields
                            continue
                        if syncf[0].type == sourcef[0].type:
                            continue
                        if str(syncf[0].name) == str(sourcef[0].name):
                            continue
                        if str(syncf[0].aliasName) == str(sourcef[0].aliasName):
                            continue
                        if syncf[0].defaultValue == sourcef[0].defaultValue:
                            continue
                        if syncf[0].length == sourcef[0].length:
                            continue
                        if syncf[0].precision == sourcef[0].precision:
                            continue
                        if syncf[0].scale == sourcef[0].scale:
                            continue
                        else:
                            print(str(field.name)+" Mismatch, building variables")
                            if field.isNullable == False: #Set the null input for AlterField
                                fNull = "NON_NULLABLE"
                            if field.isNullable == True: #Set the null input for AlterField
                                fNull = "NULLABLE"
                            if str(field.type) == "Blob": #Set the field type input for AlterField
                                fType = "Blob"
                            if str(field.type) == "Date": #Set the field type input for AlterField
                                fType = "Date"
                            if str(field.type) == "Double": #Set the field type input for AlterField
                                fType = "Double"
                            if str(field.type) == "Single": #Set the field type input for AlterField
                                fType = "Float"
                            if str(field.type) == "Small Integer": #Set the field type input for AlterField
                                fType = "Short"
                            if str(field.type) == "String": #Set the field type input for AlterField
                                fType = "Text"
                            try: #Altering target field to match source
                                if str(syncf[0].domain) != "": #If Field has domain remove domain first
                                    print("Removing domain "+str(syncf[0].domain)+" from "+str(field.name))
                                    arcpy.RemoveDomainFromField_management(os.path.join(cmdm, ds, fc), str(syncf[0].domain)) #Remove domain
                                    print("Altering " + str(field.name))
                                    arcpy.management.AlterField(str(os.path.join(cmdm, ds, fc)), str(syncf[0].name),None, str(field.aliasName), fType, str(field.length), fNull, "DO_NOT_CLEAR") #Alter field
                                    print("Reassigning domain " +str(field.domain)+ " from "+str(field.name))
                                    if str(field.domain) != "": #If WUNM has domain assigned use it on new field
                                        print("Assigning domain " + str(field.domain) + " from " + str(field.name))
                                        arcpy.management.AssignDomainToField(str(os.path.join(cmdm, ds, fc)), str(field.name),str(field.domain)) #Assigning domain
                                    else:
                                        continue
                                elif field.domain =="": #If no assigned domain continue to alter field
                                    print("Altering " + str(field.name))
                                    arcpy.management.AlterField(str(os.path.join(cmdm, ds, fc)), str(syncf[0].name),None, str(field.aliasName), fType, str(field.length), fNull, "DO_NOT_CLEAR") #Alter field
                                    if str(field.domain) != "": #If WUNM has domain assign it to the new field
                                        print("Assigning domain " + str(field.domain) + " from " + str(field.name))
                                        arcpy.management.AssignDomainToField(str(os.path.join(cmdm, ds, fc)), str(field.name),str(field.domain)) #Assigning domain
                                    else:
                                        continue
                            except Exception: #Error capture
                                print("Alter Field Error "+str(field.name))
                                e = arcpy.sys.exc_info()[1]
                                print(e.args[0])
                                arcpy.ClearWorkspaceCache_management()
                                sys.exit()
            else:
                print(str(field.name)+" not found, adding field")
                if field.isNullable == False: #Set the field Null input for AddField
                    fNull = "NON_NULLABLE"
                else:
                    fNull = "NULLABLE"
                if field.required == False: #Set the field Required input for AddField
                    fReq = "NON_REQUIRED"
                else:
                    fReq = "REQUIRED"
            if str(field.type) == "Blob": #Set the field Type input for AddField
                fType = "Blob"
            if str(field.type) == "Date": #Set the field Type input for AddField
                fType = "Date"
            if str(field.type) == "Double": #Set the field Type input for AddField
                fType = "Double"
            if str(field.type) == "Single": #Set the field Type input for AddField
                fType = "Float"
            if str(field.type) == "Small Integer": #Set the field Type input for AddField
                fType = "Short"
            if str(field.type) == "String": #Set the field Type input for AddField
                fType = "Text"
            try: #Add new field to target from source
                arcpy.management.AddField(str(os.path.join(cmdm, ds, fc)), field.name, fType, str(field.precision), str(field.scale), str(field.length), str(field.aliasName), fNull, fReq, str(field.domain))
            except Exception: #Error capture
                print("Add Field Error "+str(field.name))
                e = arcpy.sys.exc_info()[1]
                print(e.args[0])
                arcpy.ClearWorkspaceCache_management()
                sys.exit()
arcpy.ClearWorkspaceCache_management()

Time4 = ((time.time() - Time3))
print("Step 6 Complete --- %s seconds --- " % (Time4))

#Step 7 Append WUNM Database to CMDM Database
arcpy.env.workspace = r"k:\Ctown_Model\test" #Reset Workspace
for ds in dsList: #Loop through Datasets
    arcpy.env.workspace = os.path.join(wunm,ds)
    print(str(ds)+" Current Workspace in "+str(wunm))
    fcListWUNM = arcpy.ListFeatureClasses()
    fcListWUNM.remove(str(ds)+"_Net_Junctions")
    arcpy.env.workspace = os.path.join(cmdm, ds)
    fcListCMDM = arcpy.ListFeatureClasses()
    fcListCMDM.remove(str(ds)+"_Net_Junctions")
    for fc in fcListWUNM: #Loop through Featureclasses
        if fc in fcListCMDM: #Check if Featureclass exists in Database
            print(arcpy.management.GetCount(os.path.join(wunm, ds, fc)))
            if int(str(arcpy.management.GetCount(os.path.join(wunm, ds, fc))))>0:
                try:
                    print("Start Append " + str(fc))
                    arcpy.management.Append(os.path.join(wunm, ds, fc), os.path.join(cmdm, ds, fc), "NO_TEST")
                    print("Append " + str(fc) + " Success")
                except Exception:  # Error Capture
                    print("Append " + str(fc) + " Failed")
                    e = arcpy.sys.exc_info()[1]
                    print(e.args[0])
                    arcpy.ClearWorkspaceCache_management()
                    sys.exit()
            else:
                continue
arcpy.ClearWorkspaceCache_management()

Time7 = ((time.time() - Time6))
print("Step 7 Complete --- %s seconds --- " % (Time7))

tTime = ((time.time() - sTime))
print("All processes completed in %s seconds" % (tTime))
sys.exit()
