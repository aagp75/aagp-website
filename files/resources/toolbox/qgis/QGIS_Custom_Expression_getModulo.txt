from qgis.core import * # qgis module imports
from qgis.gui import * # qgis module imports

@qgsfunction(args='auto', group='Custom') #defining location of custom function
def getModulo(value1, value2, feature, parent): #defining function variables
    """ #description block
    Calculates the modulo value of the two parameters value1 and value2.
    <h2>Example usage:</h2>
    <ul>
      <li>modulo(5, 8) -> 5</li>
      <li>modulo(8, 8) -> 0</li>
      <li>modulo(8, 3) -> 2</li>
    </ul>
    """
    return value1 % value2 #python expression
